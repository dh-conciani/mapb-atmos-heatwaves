// 4. FIXED URBAN CENTROID FOR EACH CITY

function municipalityFor(meta) {
  if (!USE_MUNICIPAL_LIMITS) {
    return ee.FeatureCollection([]);
  }

  // Some historical versions store IBGE code as string, others as number.
  return municipalities.filter(
    ee.Filter.or(
      ee.Filter.eq(MUNICIPAL_CODE_FIELD, meta.ibge_code),
      ee.Filter.eq(MUNICIPAL_CODE_FIELD, parseInt(meta.ibge_code, 10))
    )
  );
}

function urbanCentroid(meta) {
  var anchor = ee.Geometry.Point([meta.lon, meta.lat]);
  var fallbackRegion = anchor.buffer(meta.radius_km * 1000);

  var muni = municipalityFor(meta);
  var region = USE_MUNICIPAL_LIMITS
    ? ee.Geometry(ee.Algorithms.If(
        muni.size().gt(0),
        muni.geometry(),
        fallbackRegion
      ))
    : fallbackRegion;

  var urban = urban2025.clip(region);

  // Area-weighted centroid of the union of urban pixels.
  var pixelArea = ee.Image.pixelArea().updateMask(urban).rename('area');
  var lon = ee.Image.pixelLonLat().select('longitude');
  var lat = ee.Image.pixelLonLat().select('latitude');

  var weighted = ee.Image.cat([
    lon.multiply(pixelArea).rename('lon_area'),
    lat.multiply(pixelArea).rename('lat_area'),
    pixelArea
  ]);

  var sums = weighted.reduceRegion({
    reducer: ee.Reducer.sum(),
    geometry: region,
    scale: 30,
    maxPixels: 1e9,
    tileScale: 4
  });

  var area = ee.Number(sums.get('area'));

  // Extremely defensive fallback: if no urban pixels are found, use anchor.
  var hasUrban = area.gt(0);
  var cx = ee.Number(ee.Algorithms.If(
    hasUrban,
    ee.Number(sums.get('lon_area')).divide(area),
    meta.lon
  ));
  var cy = ee.Number(ee.Algorithms.If(
    hasUrban,
    ee.Number(sums.get('lat_area')).divide(area),
    meta.lat
  ));

  return ee.Feature(ee.Geometry.Point([cx, cy]), {
    city_id: meta.city_id,
    city: meta.city,
    ibge_code: meta.ibge_code,
    urban_reference_year: URBAN_YEAR,
    urban_class: URBAN_CLASS,
    urban_area_m2: ee.Algorithms.If(hasUrban, area, null),
    municipality_match: muni.size().gt(0),
    centroid_method: ee.Algorithms.If(
      hasUrban,
      'area_weighted_centroid_MapBiomas_urban_2025',
      'fallback_city_anchor_no_urban_pixels'
    )
  });
}

// 5. STATION METRICS FOR AN ARBITRARY PERIOD
// ============================================================================

function stationMetrics(cityId, startDate, endDateExclusive) {
  var startMs = ee.Date(startDate).millis();
  var endMs = ee.Date(endDateExclusive).millis();

  var events = stationEvents.filter(ee.Filter.eq('city_id', cityId));

  // Allocate event DAYS to their actual dates.
  var overlapDays = events.map(function(f) {
    f = ee.Feature(f);
    var eventStart = ee.Number(f.get('start_ms'));
    var eventEndExclusive = ee.Number(f.get('end_ms')).add(DAY_MS);

    var overlapStart = eventStart.max(startMs);
    var overlapEnd = eventEndExclusive.min(endMs);

    var days = overlapEnd
      .subtract(overlapStart)
      .divide(DAY_MS)
      .max(0);

    return f.set('period_overlap_days', days);
  });

  var heatwaveDays = ee.Number(
    overlapDays.aggregate_sum('period_overlap_days')
  );

  // Whole events belong to the period in which the event ENDS.
  var ended = events
    .filter(ee.Filter.gte('end_ms', startMs))
    .filter(ee.Filter.lt('end_ms', endMs));

  var count = ended.size();

  var maxDuration = ee.Algorithms.If(
    count.gt(0),
    ended.aggregate_max('duration'),
    null
  );

  var meanDuration = ee.Algorithms.If(
    count.gt(0),
    ended.aggregate_mean('duration'),
    null
  );

  return ee.Dictionary({
    heatwave_days: heatwaveDays,
    event_count: count,
    max_event_duration: maxDuration,
    mean_event_duration: meanDuration
  });
}

// 6. SAMPLE ONE MAPBIOMAS AGGREGATED PERIOD AT ONE URBAN CENTROID
// ============================================================================

function aggregateSubset(ic, year, month) {
  return ic
    .filter(ee.Filter.eq('year', year))
    .filter(ee.Filter.eq('month', month));
}

function sampleAggregate(ic, year, month, point) {
  var subset = aggregateSubset(ic, year, month);
  var available = subset.size().gt(0);

  var values = ee.Dictionary(ee.Algorithms.If(
    available,
    ee.Image(subset.first())
      .select(METRICS)
      .reduceRegion({
        reducer: ee.Reducer.first(),
        geometry: point,
        scale: SAMPLE_SCALE,
        maxPixels: 100000,
        bestEffort: false
      }),
    ee.Dictionary({})
  ));

  return values.set('available', available);
}


// ============================================================================
// 7. BUILD ONE COMPARISON ROW
// ============================================================================

function oneComparison(cityFeature, year, month) {
  cityFeature = ee.Feature(cityFeature);
  year = ee.Number(year).toInt();
  month = ee.Number(month).toInt();

  var isAnnual = month.eq(0);

  var start = ee.Date(ee.Algorithms.If(
    isAnnual,
    ee.Date.fromYMD(year, 1, 1),
    ee.Date.fromYMD(year, month, 1)
  ));

  var end = ee.Date(ee.Algorithms.If(
    isAnnual,
    ee.Date.fromYMD(year.add(1), 1, 1),
    start.advance(1, 'month')
  ));

  var cityId = ee.String(cityFeature.get('city_id'));
  var station = stationMetrics(cityId, start, end);

  var v1s = sampleAggregate(v1, year, month, cityFeature.geometry());
  var v2s = sampleAggregate(v2, year, month, cityFeature.geometry());

  var props = ee.Dictionary({
    city_id: cityId,
    city: cityFeature.get('city'),
    ibge_code: cityFeature.get('ibge_code'),
    year: year,
    month: month,
    period_type: ee.Algorithms.If(isAnnual, 'annual', 'monthly'),
    period_start: start.format('YYYY-MM-dd'),
    period_end_exclusive: end.format('YYYY-MM-dd'),
    urban_reference_year: URBAN_YEAR,
    centroid_lon: cityFeature.geometry().coordinates().get(0),
    centroid_lat: cityFeature.geometry().coordinates().get(1),
    v1_available: v1s.get('available'),
    v2_available: v2s.get('available')
  });

  METRICS.forEach(function(metric) {
    props = props
      .set('station_' + metric, station.get(metric))
      .set('v1_' + metric, v1s.get(metric))
      .set('v2_' + metric, v2s.get(metric));
  });

  return ee.Feature(cityFeature.geometry(), props)
    .set('system:time_start', start.millis());
}

function statsFeature(fc, cityId, metric, version, periodType) {
  var observed = 'station_' + metric;
  var predicted = version + '_' + metric;

  var valid = fc
    .filter(ee.Filter.eq('city_id', cityId))
    .filter(ee.Filter.notNull([observed, predicted]))
    .map(function(f) {
      f = ee.Feature(f);
      var obs = ee.Number(f.get(observed));
      var pred = ee.Number(f.get(predicted));
      var err = pred.subtract(obs);

      return f.set({
        _error: err,
        _abs_error: err.abs(),
        _sq_error: err.pow(2)
      });
    });

  var n = valid.size();

  var corr = ee.Algorithms.If(
    n.gt(1),
    valid.reduceColumns(
      ee.Reducer.pearsonsCorrelation(),
      [observed, predicted]
    ).get('correlation'),
    null
  );

  var bias = ee.Algorithms.If(
    n.gt(0),
    valid.aggregate_mean('_error'),
    null
  );

  var mae = ee.Algorithms.If(
    n.gt(0),
    valid.aggregate_mean('_abs_error'),
    null
  );

  var rmse = ee.Algorithms.If(
    n.gt(0),
    ee.Number(valid.aggregate_mean('_sq_error')).sqrt(),
    null
  );

  return ee.Feature(null, {
    city_id: cityId,
    metric: metric,
    version: version,
    period_type: periodType,
    n: n,
    pearson_r: corr,
    bias_model_minus_station: bias,
    mae: mae,
    rmse: rmse
  });
}
